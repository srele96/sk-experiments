```
class Solution {
public:
  vector<vector<string>> partition(string s) {
    function<bool(const string&)> isPalindrome = [&](const string& str) {
      bool r = true;
      int i = 0, j = str.size() - 1;
      while (i < j) {
        if (str.at(i) != str.at(j)) {
          r = false;
        }
        ++i;
        --j;
      }
      return r;
    };

    function<vector<vector<string>>(const vector<int>&)> f = [&](auto splits) -> vector<vector<string>> {
      vector<vector<string>> r;

      int next = splits.size() == 0 ? 0 : splits[splits.size() - 1];
      for(int i = ++next; i < s.size(); ++i) {
        vector<int> p = splits; //
        p.push_back(i);
        for(auto row : f(p)) {
          r.push_back(row);
        }
      }

      vector<string> strings;
      bool allPalindromes = true;
      int prevIdx = 0;
      for(auto i : splits) {
        const string str = s.substr(prevIdx, i - prevIdx);
        if (!isPalindrome(str)) allPalindromes = false;
        strings.emplace_back(str);
        // cout << i << ", ";
        // cout << s.substr(prevIdx, i - prevIdx) << ", ";
        prevIdx = i;
      }
      const string str = s.substr(prevIdx, s.size() - prevIdx);
      if (!isPalindrome(str)) allPalindromes = false;
      strings.emplace_back(str);

      // cout << allPalindromes << ", ";
      // for(auto v : strings) {
      //   cout << v << ", ";
      // }

      // cout << "\n";

      if (allPalindromes) r.emplace_back(strings);

      return r;
    };

    // for(auto r : f({})) {
    //   for(auto s : r) {
    //     cout << s << ", ";
    //   }
    //   cout << "\n";
    // }

    return f({});
  }
};
```
